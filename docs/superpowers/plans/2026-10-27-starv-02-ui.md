# starV 計畫 2：頁面與基礎 UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 讓 starV 變成可以在瀏覽器裡玩的頁面：建置登記、i18n、閃耀舞台風樣式、2D 頭像、存檔與匯出入、主迴圈與離線報告，以及「藝人／工作／招募／公司」四個分頁與日誌。

**Architecture:** 計畫 1 的純邏輯核心不動 DOM；本計畫在上面加三層：`starv_store.js`（存檔，node 可測）、`starv_avatar.js`（SVG 字串，node 可測）、`starv_ui.js`（DOM 工具、彈窗佇列、分頁登記）。每個分頁是一個獨立檔案 `starv_view_<id>.js`，用 `PriTestStarvUI.register()` 登記；日誌與離線報告在 `starv_log.js`；`starv.js` 最後載入，負責啟動、主迴圈、頂部列與分頁列。企劃、業界兩個分頁在計畫 3、4、6 才加入（本計畫不放空殼）。

**Tech Stack:** Vanilla ES5、Python 建置（`generate.py`）、node（單元測試）、Playwright（`tools/midnight_check/node_modules` 既有安裝）。

## Global Constraints

- CLAUDE.md §5.2.1：樣式放 `static_src/css/starv.css`，class／id／keyframes 前綴 `sv-`；JS 放 `static_src/starv*.js`、命名空間 `window.PriTestStarv*`；字串鍵前綴 `sv_`，放 `site_src/i18n_starv.py`

- 手機直向優先：16px 側邊留白、不得出現水平捲動
- `prefers-reduced-motion: reduce` 時關閉裝飾動畫

- 離線補算 ≥ 3 天才顯示離線報告
- Playwright 一律用 `page.dispatchEvent(selector, "click")` 觸發（CLAUDE.md §4.6 的經驗；這頁每秒重畫）

```js
// 明星經紀 starV：存檔（規格 §17）。
// localStorage["pritest-starv-save"] 存整個 state JSON；每次寫入前把舊的搬到 -backup，主存檔壞掉時讀備份。
// 匯出字串＝"SV1." + base64url(UTF-8 JSON)，給換裝置用。讀到的存檔一律經過 Core.migrate 補齊欄位。
(function () {
  "use strict";

  var root = typeof window !== "undefined" ? window : globalThis;
  var Core = root.PriTestStarvCore;
  var KEY = "pritest-starv-save";
  var BACKUP = KEY + "-backup";
  var PREFIX = "SV1.";

  function defaultStorage() {
    try { return root.localStorage || null; } catch (e) { return null; }
  }
  function valid(st) {
    return !!st && typeof st === "object" && typeof st.version === "number" &&
      !!st.company && typeof st.company === "object" && Array.isArray(st.artists);
  }
  function parse(text) {
    var st = JSON.parse(text);
    if (!valid(st)) throw new Error("starV: 存檔格式不正確");
    return Core.migrate(st);
  }

  function load(storage) {
    storage = storage || defaultStorage();
    if (!storage) return null;
    var keys = [KEY, BACKUP];
    for (var i = 0; i < keys.length; i++) {
      try {
        var text = storage.getItem(keys[i]);
        if (text) return parse(text);
      } catch (e) { /* 這份壞了，試下一份 */ }
    }
    return null;
  }

  function save(st, storage) {
    storage = storage || defaultStorage();
    if (!storage) return false;
    try {
      var prev = storage.getItem(KEY);
      if (prev) storage.setItem(BACKUP, prev);
      storage.setItem(KEY, JSON.stringify(st));
      return true;
    } catch (e) { return false; }   // 私密模式或容量不足：只是記不住
  }

  function clear(storage) {
    storage = storage || defaultStorage();
    if (!storage) return;
    try { storage.removeItem(KEY); storage.removeItem(BACKUP); } catch (e) { /* 忽略 */ }
  }

  function toB64(str) { return root.btoa(unescape(encodeURIComponent(str))); }
  function fromB64(b64) { return decodeURIComponent(escape(root.atob(b64))); }

  function encode(st) {
    return PREFIX + toB64(JSON.stringify(st)).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
  }

  function decode(code) {
    var s = String(code || "").replace(/\s+/g, "");
    if (s.indexOf(PREFIX) !== 0) throw new Error("starV: 不是 starV 的存檔字串");
    var body = s.slice(PREFIX.length);
    if (!body || !/^[A-Za-z0-9_-]+$/.test(body)) throw new Error("starV: 存檔字串含有無效字元");
    var b64 = body.replace(/-/g, "+").replace(/_/g, "/");
    while (b64.length % 4) b64 += "=";
    return parse(fromB64(b64));
  }

  root.PriTestStarvStore = { KEY: KEY, BACKUP: BACKUP, load: load, save: save, clear: clear, encode: encode, decode: decode };
})();
```

```js
    Object.keys(attrs || {}).forEach(function (k) {
      var v = attrs[k];
      if (v === null || v === undefined || v === false) return;
      if (k === "text") node.textContent = v;
      else if (k === "class") node.className = v;
      else if (k === "style") node.setAttribute("style", v);
      else if (k === "value") node.value = v;
      else if (k.indexOf("on") === 0) node.addEventListener(k.slice(2), v);
      else if (v === true) node.setAttribute(k, "");
      else node.setAttribute(k, v);
    });
    (children || []).forEach(function (c) {
      if (c === null || c === undefined || c === false) return;
      node.appendChild(typeof c === "string" || typeof c === "number" ? document.createTextNode(String(c)) : c);
    });
    return node;
```

- [ ] **Step 4：建立 `static_src/starv_view_gacha.js`**

```js
// 明星經紀 starV：「招募」分頁（兩個抽卡池、PICKUP、保底、候補名單、翻卡演出；規格 §5.1、§15.2）。
// 抽卡結果先存檔再演出，避免演出途中關頁面造成結果遺失。新 SSR 在翻卡之後接全螢幕介紹。
(function () {
  "use strict";

  var Core = window.PriTestStarvCore;
  var Roster = window.PriTestStarvRoster;
  var UI = window.PriTestStarvUI;
  var t = UI.t, el = UI.el;
  var FLIP_MS = 220;

  function findAny(st, id) {
    var a = Core.artistById(st, id);
    if (a) return a;
    for (var i = 0; i < st.waitlist.length; i++) if (st.waitlist[i].artist.id === id) return st.waitlist[i].artist;
    return null;
  }
  function pct(x) { return Math.round(x * 1000) / 10; }

  function pickupBanner(st) {
    var items = ["SSR", "SR"].map(function (r) { return Roster.byId(st.gacha.pickup[r]); }).filter(Boolean);
    if (!items.length) return null;
    return el("div", { class: "sv-card sv-pickup" }, [
      el("div", { class: "sv-pickup-label", text: t("sv_gacha_pickup") }),
      el("div", { class: "sv-pickup-row" }, items.map(function (e) {
        return el("div", { class: "sv-pickup-item" }, [
          UI.avatar(e, 64),
          el("div", {}, [
            UI.rarity(e.rarity),
            el("div", { class: "sv-pickup-name", text: UI.L(e.name) }),
            el("div", { class: "sv-small sv-muted", text: UI.dirIcon(e.dir) + " " + UI.dirName(e.dir) })
          ])
        ]);
      }))
    ]);
  }

  function doPull(App, poolId, count) {
    var res = App.act(function (st) { return Core.pull(st, poolId, count); });
    if (res.err) { UI.toast(t("sv_err_" + res.err), "bad"); return; }
    App.saveNow();
    openReveal(App, res.results);
  }

  function pool(st, App, id) {
    var r = Core.rates(st, id);
    var fmt = function (c) { return id === "money" ? UI.money(c) : t("sv_tickets", { n: c }); };
    return el("div", { class: "sv-card sv-pool sv-pool-" + id }, [
      el("h3", { text: t("sv_pool_" + id) }),
      el("p", { class: "sv-muted sv-small", text: t("sv_pool_" + id + "_desc") }),
      el("div", { class: "sv-small", text: t("sv_pool_rates", { sr: pct(r.SR), ssr: pct(r.SSR) }) }),
      id === "money" ? el("div", { class: "sv-small sv-muted", text: t("sv_daily_left", { n: Core.dailyPullsLeft(st) }) }) : null,
      el("div", { class: "sv-row" }, [
        UI.btn(t("sv_pull_1", { cost: fmt(Core.pullCost(st, id, 1)) }), function () { doPull(App, id, 1); }, "sv-btn-ghost", { "data-act": "pull-" + id + "-1" }),
        UI.btn(t("sv_pull_10", { cost: fmt(Core.pullCost(st, id, 10)) }), function () { doPull(App, id, 10); }, "sv-btn-primary", { "data-act": "pull-" + id + "-10" })
      ]),
      el("div", { class: "sv-small sv-muted", text: t("sv_pull_10_note") })
    ]);
  }

  function resultNote(st, r) {
    if (r.lost) return t("sv_res_lost");
    if (r.dup === "limitBreak") { var a = findAny(st, r.artistId); return t("sv_res_lb", { n: a ? a.limitBreak : 1 }); }
    if (r.dup === "tickets") return t("sv_res_tickets", { n: r.tickets });
    if (r.waitlisted) return t("sv_res_wait");
    return "";
  }

  // 流失的人不在 state 裡：SR/SSR 用名錄資料畫，R 畫剪影
  function cardFace(st, r) {
    var a = findAny(st, r.artistId), e = r.rosterId ? Roster.byId(r.rosterId) : null;
    var who = a || e;
    return el("div", { class: "sv-gcard-front" }, [
      who ? UI.avatar(who, 72) : el("span", { class: "sv-ava sv-ava-r sv-ava-ghost", style: "--sv-size:72px" }),
      UI.rarity(r.rarity),
      el("div", { class: "sv-gcard-name", text: a ? UI.name(a) : e ? UI.L(e.name) : "???" }),
      who ? el("div", { class: "sv-small sv-muted", text: UI.dirIcon(who.dir) + " " + UI.dirName(who.dir) }) : null,
      el("div", { class: "sv-gcard-note", text: resultNote(st, r) })
    ]);
  }

  function openReveal(App, results) {
    var st = App.st();
    UI.openModal(function (box, close) {
      var cards = results.map(function (r) {
        return el("div", { class: "sv-gcard sv-gcard-" + r.rarity.toLowerCase() }, [
          el("div", { class: "sv-gcard-in" }, [el("div", { class: "sv-gcard-back", text: "★" }), cardFace(st, r)])
        ]);
      });
      var idx = 0, timer = null;
      var skipBtn = UI.btn(t("sv_reveal_skip"), finish, "sv-btn-ghost", { "data-act": "reveal-skip" });
      var doneBtn = UI.btn(t("sv_reveal_done"), close, "sv-btn-primary", { "data-act": "reveal-done", hidden: true });
      function flipNext() {
        if (idx >= cards.length) { finish(); return; }
        cards[idx++].classList.add("sv-flipped");
        timer = setTimeout(flipNext, FLIP_MS);
      }
      function finish() {
        clearTimeout(timer);
        while (idx < cards.length) cards[idx++].classList.add("sv-flipped");
        skipBtn.hidden = true;
        doneBtn.hidden = false;
      }
      box.appendChild(el("h2", { class: "sv-modal-title", text: t("sv_reveal_title") }));
      box.appendChild(el("div", { class: "sv-gacha-grid" + (results.length === 1 ? " sv-gacha-one" : "") }, cards));
      box.appendChild(el("div", { class: "sv-row sv-row-end" }, [skipBtn, doneBtn]));
      timer = setTimeout(flipNext, 300);
    }, { cls: "sv-modal-wide sv-reveal", locked: true, onClose: function () { spotlights(App, results); } });
  }

  function spotlights(App, results) {
    var st = App.st();
    results.forEach(function (r) {
      if (r.rarity !== "SSR" || r.dup || r.lost) return;
      var a = findAny(st, r.artistId), e = Roster.byId(r.rosterId);
      if (!a || !e) return;
      UI.openModal(function (box, close) {
        box.appendChild(el("div", { class: "sv-spot" }, [
          el("div", { class: "sv-spot-rays" }),
          UI.avatar(a, 160),
          el("div", {}, [UI.rarity("SSR")]),
          el("h2", { class: "sv-spot-name", text: UI.L(e.name) }),
          el("div", { class: "sv-muted", text: UI.dirIcon(e.dir) + " " + UI.dirName(e.dir) + "・" + t("sv_age", { n: e.age }) }),
          el("p", { class: "sv-spot-bio", text: UI.L(e.bio) }),
          e.skill ? el("div", { class: "sv-skill" }, [
            el("b", { text: t("sv_spot_skill") + "：" + UI.L(e.skill.name) }),
            el("div", { class: "sv-small", text: UI.L(e.skill.desc) })
          ]) : null,
          UI.btn(t("sv_reveal_done"), close, "sv-btn-primary", { "data-act": "spot-done" })
        ]));
      }, { cls: "sv-modal-spot" });
    });
  }

  function waitlist(st, App) {
    var rows = st.waitlist.map(function (w, i) {
      var a = w.artist;
      return el("div", { class: "sv-wait-row" }, [
        UI.avatar(a, 44),
        el("div", { class: "sv-grow" }, [
          el("div", {}, [UI.rarity(a.rarity), " ", el("b", { text: UI.name(a) })]),
          el("div", { class: "sv-small sv-muted", text: UI.dirIcon(a.dir) + " " + UI.dirName(a.dir) + "・" + t("sv_wait_left", { n: w.until - st.day }) })
        ]),
        UI.btn(t("sv_wait_accept"), function () {
          var err = App.act(function (s) { return Core.acceptWaitlist(s, i); });
          if (err) UI.toast(t("sv_err_" + err), "bad");
        }, "sv-btn-gold", { "data-act": "wait-accept" })
      ]);
    });
    return el("div", { class: "sv-card" }, [
      el("h3", { text: t("sv_wait_title") }),
      el("p", { class: "sv-small sv-muted", text: t("sv_wait_hint") }),
      rows.length ? el("div", {}, rows) : el("p", { class: "sv-muted", text: t("sv_wait_empty") })
    ]);
  }

  UI.register({
    id: "gacha", order: 30, icon: "✨", labelKey: "sv_tab_gacha",
    render: function (c, App) {
      var st = App.st();
      var banner = pickupBanner(st);
      if (banner) c.appendChild(banner);
      c.appendChild(el("div", { class: "sv-card sv-pity", text: t("sv_gacha_pity", { n: Core.pityLeft(st) }) }));
      c.appendChild(el("div", { class: "sv-pools" }, [pool(st, App, "ticket"), pool(st, App, "money")]));
      c.appendChild(waitlist(st, App));
    }
  });
})();
```

```js
// 明星經紀 starV：「工作」分頁（企劃書風格的看板、選角、進行中工作、組合；規格 §6.2–§6.4、§15.2）。

(function () {

  function jobCard(st, job, onclick) {

          job.slots > 1 ? t("sv_job_slots", { n: job.slots }) : null].filter(Boolean).join("・") }),
        el("div", { class: "sv-job-pay", text: UI.money(job.pay) }),
        el("div", { class: "sv-small sv-muted", text: t("sv_job_req", { fame: UI.num(job.minFame), prof: job.minProf }) +

        job.isWork ? el("span", { class: "sv-chip sv-chip-work", text: t("sv_job_work") }) : null

  function openCasting(App, jobId) {
    var sel = [];
    UI.openModal(function (box, close) {
      var st = App.st(), job = null;
      st.board.forEach(function (j) { if (j.id === jobId) job = j; });
      if (!job) { close(); UI.toast(t("sv_err_noJob"), "bad"); return; }
      sel = sel.filter(function (id) { var a = Core.artistById(st, id); return a && !Core.eligible(st, a, job); });
      box.appendChild(UI.modalHead(t("sv_cast_title"), close));
      box.appendChild(jobCard(st, job, null));
      var rows = st.artists.map(function (a) {
        return { a: a, err: Core.eligible(st, a, job), est: Core.estimate(st, [a.id], job) };
      }).sort(function (x, y) { return (x.err ? 1 : 0) - (y.err ? 1 : 0) || y.est.ratio - x.est.ratio; });
      box.appendChild(el("p", { class: "sv-small", text: t("sv_cast_pick", { n: job.slots, k: sel.length }) }));
      box.appendChild(el("div", { class: "sv-cast-list" }, rows.map(function (r) {
        var on = sel.indexOf(r.a.id) >= 0;

          class: "sv-cast-row" + (on ? " sv-sel" : "") + (r.err ? " sv-dis" : ""), "data-aid": r.a.id, role: "button", tabindex: "0",

            var i = sel.indexOf(r.a.id);

            el("div", { text: t("sv_est", { label: t("sv_est_" + r.est.label) }) }),
            el("div", { class: "sv-stars", text: fitStars(r.est.ratio) })

      }, "sv-btn-primary", { "data-act": "group-create" })])

  }

  UI.register({
    id: "jobs", order: 20, icon: "🎬", labelKey: "sv_tab_jobs",
```

```js
// 明星經紀 starV：日誌抽屜、日誌文字、離線報告（規格 §2、§15.2）。
// 日誌項目是 { seq, day, type, data }（Core 產生），這裡依 type 組出 sv_log_<type> 的參數。
(function () {
  "use strict";

  var D = window.PriTestStarvData;
  var Core = window.PriTestStarvCore;
  var UI = window.PriTestStarvUI;
  var t = UI.t, el = UI.el;
  var drawer = { open: false, aid: "" };
  var REPORT_LINES = 20;

  function artistLabel(st, id) { var a = Core.artistById(st, id); return a ? UI.name(a) : "—"; }
  // 已經不在公司的人（候補流失、解約）由日誌裡的 rosterId／nameSeed 還原名字
  function who(st, d) {
    if (d.aid) return artistLabel(st, d.aid);
    if (d.rosterId || d.nameSeed != null) return Core.artistName({ rosterId: d.rosterId || null, nameSeed: d.nameSeed || 0, gender: d.gender || "f" }, UI.lang());
    return "—";
  }

  function text(st, e) {
    var d = e.data || {}, p = { name: who(st, d) };
    if (e.type === "dup_lb") p.lb = d.lb;
    else if (e.type === "dup_ticket") p.n = d.tickets;
    else if (e.type === "course_done") {
      var c = D.COURSE_BY_ID[d.courseId];
      p.course = UI.courseName(d.courseId); p.stat = c ? UI.statName(c.stat) : ""; p.gain = d.gain;
    } else if (e.type === "promo_done") p.fame = d.fame;
    else if (e.type === "job_done") {
      p.names = (d.aids || []).map(function (id) { return artistLabel(st, id); }).join("、");
      p.title = Core.jobTitle({ dir: d.dir, title: d.title }, UI.lang());
      p.result = t("sv_result_" + d.result);
      p.pay = UI.money(d.pay);
      p.stars = new Array((d.stars || 0) + 1).join("★");
    } else if (e.type === "minor") {
      var ev = null;
      D.MINOR_EVENTS.forEach(function (x) { if (x.id === d.ev) ev = x; });
      p.ev = ev ? UI.L(ev.text) : String(d.ev);
    } else if (e.type === "level_up") p.lv = d.lv;
    else if (e.type === "month_end") { p.salary = UI.money(d.salary); p.upkeep = UI.money(d.upkeep); p.div = UI.money(d.dividends); }
    else if (e.type === "birthday_year") p.year = d.year;
    return t("sv_log_" + e.type, p);
  }

  function involves(e, aid) { var d = e.data || {}; return d.aid === aid || (!!d.aids && d.aids.indexOf(aid) >= 0); }
  function line(st, e) {
    return el("li", { class: "sv-log-line sv-log-" + e.type }, [
      el("span", { class: "sv-log-day", text: UI.dateText(e.day) }),
      el("span", { text: text(st, e) })
    ]);
  }

  function drawerNode(App) {
    var st = App.st();
    var toggle = el("button", {
      type: "button", class: "sv-log-btn" + (drawer.open ? " sv-on" : ""), "data-act": "log-toggle", text: "📜 " + t("sv_log"),
      onclick: function () { drawer.open = !drawer.open; App.render(); }
    });
    if (!drawer.open) return el("div", { class: "sv-log-wrap" }, [toggle]);
    if (drawer.aid && !Core.artistById(st, drawer.aid)) drawer.aid = "";
    var logs = st.log.slice().reverse().filter(function (e) { return !drawer.aid || involves(e, drawer.aid); });
    var filter = UI.select([{ value: "", label: t("sv_log_all") }].concat(st.artists.map(function (a) { return { value: a.id, label: UI.name(a) }; })),
      drawer.aid, function (v) { drawer.aid = v; App.render(); }, { "data-act": "log-filter" });
    return el("div", { class: "sv-log-wrap" }, [
      toggle,
      el("aside", { class: "sv-log-panel" }, [
        el("div", { class: "sv-row" }, [el("b", { class: "sv-grow", text: t("sv_log") }), filter]),
        logs.length ? el("ul", { class: "sv-log-list" }, logs.map(function (e) { return line(st, e); }))
          : el("p", { class: "sv-muted", text: t("sv_log_empty") })
      ])
    ]);
  }

  function signed(v, fmt) { return (v >= 0 ? "+" : "−") + fmt(Math.abs(v)); }

  function report(App, rep) {
    UI.openModal(function (box, close) {
      var st = App.st(), cnt = { great: 0, success: 0, normal: 0, fail: 0 }, n = 0;
      rep.logs.forEach(function (e) { if (e.type === "job_done") { n++; cnt[e.data.result]++; } });
      box.appendChild(UI.modalHead(t("sv_report_title", { days: rep.days }), close));
      if (rep.capped) box.appendChild(el("p", { class: "sv-warn", text: t("sv_report_capped") }));
      box.appendChild(el("ul", { class: "sv-report-list" }, [
        el("li", { text: t("sv_report_money", { v: signed(rep.money, UI.money) }) }),
        el("li", { text: t("sv_report_tickets", { v: signed(rep.tickets, String) }) }),
        el("li", { text: t("sv_report_rep", { v: signed(rep.rep, UI.num) }) }),
        rep.levelTo !== rep.levelFrom ? el("li", { class: "sv-good", text: t("sv_report_level", { from: rep.levelFrom, to: rep.levelTo }) }) : null,
        el("li", { text: t("sv_report_jobs", { n: n, g: cnt.great, s: cnt.success, o: cnt.normal, f: cnt.fail }) })
      ]));
      box.appendChild(el("h3", { text: t("sv_report_recent") }));
      box.appendChild(el("ul", { class: "sv-log-list" }, rep.logs.slice(-REPORT_LINES).reverse().map(function (e) { return line(st, e); })));

    }, { cls: "sv-report" });
  }

  UI.addExtra(drawerNode);
```

```js
      UI.bar(next === null ? 1 : c.rep, next === null ? 1 : next, "sv-bar-sat"),

  function saveCard(st, App) {
    var importBox = el("textarea", { class: "sv-textarea", placeholder: t("sv_save_import_ph"), "data-act": "save-import-text", value: ui.importText,
      oninput: function () { ui.importText = importBox.value; ui.pending = null; } });
    var rows = [
      el("h3", { text: t("sv_save") }),
      el("p", { class: "sv-small sv-muted", text: t("sv_save_hint") }),
      el("div", { class: "sv-row" }, [UI.btn(t("sv_save_export"), function () { ui.code = Store.encode(App.st()); App.render(); }, "", { "data-act": "save-export" })])
    ];
    if (ui.code) {
      rows.push(el("textarea", { class: "sv-textarea", readonly: true, "data-act": "save-code", value: ui.code }));
      rows.push(el("div", { class: "sv-row" }, [UI.btn(t("sv_save_copy"), function () {
        try { navigator.clipboard.writeText(ui.code).then(function () { UI.toast(t("sv_save_copied")); }); } catch (e) { /* 不支援就手動複製 */ }
      }, "sv-btn-ghost", { "data-act": "save-copy" })]));
    }
    rows.push(importBox);
    rows.push(el("div", { class: "sv-row" }, [
      UI.btn(t("sv_save_import"), function () {
        try { ui.pending = Store.decode(ui.importText); } catch (e) { ui.pending = null; UI.toast(t("sv_err_import"), "bad"); }
        App.render();
      }, "", { "data-act": "save-import" }),
      ui.pending ? UI.btn(t("sv_save_import_confirm"), function () {
        var next = ui.pending;
        ui.pending = null; ui.importText = ""; ui.code = "";
        App.replaceState(next);
        UI.toast(t("sv_save_imported"), "gold");
      }, "sv-btn-danger", { "data-act": "save-import-confirm" }) : null
    ]));
    rows.push(el("div", { class: "sv-row sv-row-end" }, [
      ui.confirmReset
        ? UI.btn(t("sv_save_reset_confirm"), function () { App.reset(); }, "sv-btn-danger", { "data-act": "save-reset-confirm" })
        : UI.btn(t("sv_save_reset"), function () { ui.confirmReset = true; App.render(); }, "sv-btn-ghost", { "data-act": "save-reset" })
    ]));
    return el("div", { class: "sv-card" }, rows);
  }

  function debugCard(st, App) {
    return el("div", { class: "sv-card sv-debug" }, [
      el("h3", { text: t("sv_debug") }),
      el("div", { class: "sv-row" }, [el("span", { text: t("sv_debug_speed") })].concat([1, 10, 60].map(function (n) {
        return UI.btn("×" + n, function () { App.setSpeed(n); }, App.speed() === n ? "sv-btn-primary" : "sv-btn-ghost", { "data-act": "debug-speed-" + n });
      }))),
      el("div", { class: "sv-row" }, [
        UI.btn(t("sv_debug_money"), function () { App.act(function (s) { s.company.money += 100000; }); }, "", { "data-act": "debug-money" }),
        UI.btn(t("sv_debug_tickets"), function () { App.act(function (s) { s.company.tickets += 10; }); }, "", { "data-act": "debug-tickets" }),
        UI.btn(t("sv_debug_skip", { n: 7 }), function () { App.skip(7); }, "", { "data-act": "debug-skip-7" }),
        UI.btn(t("sv_debug_skip", { n: 30 }), function () { App.skip(30); }, "", { "data-act": "debug-skip-30" })
      ])
    ]);
  }

  UI.register({
    id: "company", order: 90, icon: "🏢", labelKey: "sv_tab_company",
    render: function (c, App) {
      var st = App.st();
      c.appendChild(infoCard(st, App));
      c.appendChild(saveCard(st, App));
      if (App.debug) c.appendChild(debugCard(st, App));
    }
  });
})();
```

```js
step("三種語言：各分頁沒有漏翻、手機寬度沒有水平捲動", async function (page) {
  var tabs = ["artists", "jobs", "gacha", "company"];
  var langs = ["ja", "en", "zh"];
  for (var i = 0; i < langs.length; i++) {
    await page.dispatchEvent('.lang-btn[data-lang="' + langs[i] + '"]', "click");
    for (var j = 0; j < tabs.length; j++) {
      await tab(page, tabs[j]);
      await noRawKeys(page, langs[i] + " / " + tabs[j]);
      var over = await page.evaluate(function () { return document.documentElement.scrollWidth - window.innerWidth; });
      assert.ok(over <= 0, langs[i] + " / " + tabs[j] + " 水平溢出 " + over + "px");
      if (langs[i] === "zh") await shot(page, "tab-" + tabs[j]);
    }
  }
  var aid = (await st(page)).artists[0].id;
  await tab(page, "artists");
  await page.dispatchEvent('.sv-acard[data-aid="' + aid + '"]', "click");
  await page.waitForSelector(".sv-modal-detail");
  await shot(page, "artist-detail");
  await act(page, "modal-close");
});
```
